xx <- fips_rates %>%  
  filter(
    !is.na(rates), 
    !is.na(fips)
  ) %>% 
  group_by(fips, race) %>% 
  summarise(
    n = n(),
    max = max(rates, na.rm = T), 
    min = min(rates, na.rm = T), 
    mean = mean(rates, na.rm = T), 
    pct_diff = max / min - 1 
  )

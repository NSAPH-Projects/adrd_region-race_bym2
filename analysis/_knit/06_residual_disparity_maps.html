pos_disparity_sf <- county_sf %>% 
  mutate(
    county = as.integer(GEOID)
  ) %>% 
  select(county) %>% 
  right_join(pos_disparity)
## Joining, by = "county"
pos_disparity_sf %>% 
  ggplot() +
  geom_sf(aes(fill = total_disparity), lwd = 0.2) + 
  theme(legend.position = "bottom")
